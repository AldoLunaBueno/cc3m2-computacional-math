\begin{frame}{Trust Region: Formulación del Enfoque}
    \begin{block}{Criterio de optimización local}
        \small
        La actualización del punto \(x_k\) exige definir la \textbf{dirección} del paso y su \textbf{magnitud}.
    \end{block}

    \vspace{0.5em}
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Búsqueda Lineal}
                \scriptsize
                \begin{enumerate}
                    \item Se calcula la dirección de descenso \(\mathbf{d}\).
                    \item Se determina la tasa de paso \(\alpha\).
                \end{enumerate}
                \vspace{0.3em}
                La dirección elegida es fija e independiente del tamaño final del paso.
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Región de Confianza}
                \scriptsize
                \begin{enumerate}
                    \item Se delimita un radio máximo de validez (\(\Delta\)).
                    \item Se resuelve la dirección y paso simultáneamente dentro de la cota.
                \end{enumerate}
                \vspace{0.3em}
                El vector de paso se ajusta dinámicamente a la restricción impuesta.
            \end{alertblock}
        \end{column}
    \end{columns}

    \vspace{0.6em}
    \footnotesize
    \begin{block}{Principio de aproximación}
        \scriptsize
        La aproximación local es precisa en una vecindad inmediata de \(x_k\), pero pierde validez al alejarse. El radio \(\Delta\) cuantifica el dominio en el que el modelo predice adecuadamente la función objetivo.
    \end{block}
\end{frame}

\begin{frame}{Construcción del Modelo Cuadrático}
    \begin{block}{Aproximación de Taylor de segundo orden}
        \small
        \[
            \hat f(x_k + p) \;=\;
            f(x_k) \;+\;
            f'(x_k)\,p \;+\;
            \tfrac{1}{2}f''(x_k)\,p^{2}
        \]
    \end{block}

    \vspace{-0.2em}
    \begin{center}
    \begin{tikzpicture}[x=1.45cm, y=1.45cm, font=\scriptsize]

    \vspace{-0.3em}

\end{frame}

\begin{frame}{Resolución del Subproblema}
    \begin{block}{Subproblema con restricción de norma}
        \small
        \[
            p_k \;=\; \arg\min_{|p|\,\le\,\Delta}\ \hat f(x_k+p)
        \]
        El mínimo global no acotado corresponde al paso de Newton \(\;p^{*} = -f'(x_k)/f''(x_k)\). Se presentan dos casos posibles:
    \end{block}

    \vspace{0.3em}
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Caso interior: \(|p^{*}| \le \Delta\)}
                \scriptsize
                Se adopta \(p = p^{*}\).\\
                La restricción no está activa: la actualización coincide con el paso de Newton estándar.
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Caso frontera: \(|p^{*}| > \Delta\)}
                \scriptsize
                Se aplica la cota: \(p = \operatorname{sign}(p^{*})\Delta\).\\
                La restricción está activa: el paso óptimo excede el radio admisible.
            \end{alertblock}
        \end{column}
    \end{columns}

    \vspace{0.6em}
    \begin{center}

    \end{center}
\end{frame}

\begin{frame}{Evaluación del Modelo: Razón de Acuerdo \(\rho\)}
    \begin{block}{Criterio de reducción relativa}
        \small
        \[
            \rho \;=\;
            \frac{f(x_k) - f(x_k+p)}{f(x_k) - \hat f(x_k+p)}
            \;=\;
            \frac{\text{Reducción Real}}{\text{Reducción Predicha}}
        \]
        Dado que el denominador es estrictamente positivo en la dirección de descenso, la magnitud de \(\rho\) determina la concordancia del modelo cuadrático.
    \end{block}

    \vspace{0.4em}
    \begin{center}
    \scriptsize
    \begin{tabular}{c l l}
        \toprule
        Rango de \(\rho\) & Concordancia & Criterio de actualización \\
        \midrule
        \(\rho < 0\)                   & Divergencia (\(f\) se incrementa) & Rechazar paso; \(\Delta \gets \Delta/2\) \\
        \(0 \le \rho < 0.25\)          & Discrepancia alta               & Rechazar paso; \(\Delta \gets \Delta/2\) \\
        \(0.25 \le \rho \le 0.75\)     & Concordancia aceptable          & Aceptar paso; \(\Delta\) constante \\
        \(\rho > 0.75\) y \(|p|=\Delta\) & Concordancia alta               & Aceptar paso; \(\Delta \gets 2\Delta\) \\
        \bottomrule
    \end{tabular}
    \end{center}

    \vspace{0.5em}
    \footnotesize
    \begin{alertblock}{Mecanismo de conservación}
        \scriptsize
        Ante un paso rechazado, la posición permanece invariable: \(x_{k+1} = x_k\). La evaluación computada permite reducir el radio \(\Delta\) garantizando que la función objetivo no empeore de una iteración a otra.
    \end{alertblock}
\end{frame}

\begin{frame}{Pseudocódigo}
    \small
    \linespread{0.85}\selectfont
    \setlength{\lineskip}{0pt}
    \begin{algorithmic}[1]
        \Require \(f,\ f',\ f''\), valor inicial \(x\), radio \(\Delta\), tolerancia \(\epsilon\)
        \For{\(k = 1\) \textbf{to} \(k_{\max}\)}
            \State \(p^{*} \gets -f'(x)/f''(x)\) \Comment{Mínimo libre no acotado}
            \State \(p \gets \operatorname{sign}(p^{*}) \cdot \min\big(|p^{*}|,\ \Delta\big)\) \Comment{Proyección al dominio admisible}
            \State \(\hat f \gets f(x) + f'(x)\,p + \tfrac{1}{2}f''(x)\,p^{2}\)
            \State \(\rho \gets \big(f(x) - f(x+p)\big) \,/\, \big(f(x) - \hat f\,\big)\)
            \If{\(\rho < 0.25\)}
                \State \(\Delta \gets \Delta/2\) \Comment{Rechazo: \(x\) permanece constante}
            \Else
                \State \(x \gets x + p\) \Comment{Paso aceptado}
                \If{\(\rho > 0.75\) \textbf{y} \(|p| = \Delta\)}
                    \State \(\Delta \gets 2\Delta\) \Comment{Expansión del radio}
                \EndIf
            \EndIf
            \If{\(|f'(x)| < \epsilon\)}
                \State \textbf{break}
            \EndIf
        \EndFor
        \State \Return \(x\)
    \end{algorithmic}
\end{frame}

    \vspace{0.6em}
    \begin{block}{Observaciones sobre el comportamiento del método}
        \footnotesize
        \begin{enumerate}
            \item \textbf{Cota de desplazamiento:} Previene divergencias por curvaturas bajas (\(f'' \approx 0\)) donde Newton propone magnitudes desproporcionadas.
            \item \textbf{Control de discrepancia:} El parámetro \(\rho\) detecta inconsistencias entre predicción y valor real, impidiendo avances infructuosos.
            \item \textbf{Autoajuste paramétrico:} El radio \(\Delta\) se actualiza de manera reactiva en función de la no-linealidad local encontrada.
        \end{enumerate}

\begin{frame}[fragile]{Implementación (1D)}
\begin{lstlisting}[language=Python, basicstyle=\ttfamily\tiny]
from math import exp

def trust_region_1d(f, df, d2f, x, delta=1.0, eps=1e-8, kmax=50):
    for _ in range(kmax):
        fx, g, h = f(x), df(x), d2f(x)
        p = -g / h                                  # Paso de Newton libre
        if abs(p) > delta:                          # Restriccion activa
            p = delta if p > 0 else -delta
        m = fx + g*p + 0.5*h*p*p                    # Reduccion predicha
        rho = (fx - f(x + p)) / (fx - m)            # Razon de acuerdo
        if rho < 0.25:
            delta /= 2                              # Rechazo del paso
        else:
            x = x + p                               # Aceptacion del paso
            if rho > 0.75 and abs(p) == delta:
                delta *= 2                          # Expansion de la region
        if abs(df(x)) < eps:
            break
    return x

f   = lambda x: exp(x-2) - x
df  = lambda x: exp(x-2) - 1
d2f = lambda x: exp(x-2)
# trust_region_1d(f, df, d2f, 0.0, 1.0) -> x = 2.0
\end{lstlisting}
\end{frame}

\begin{frame}{Análisis de Convergencia}
    \begin{alertblock}{Propiedades teóricas}
        \footnotesize
        \textbf{Convergencia global:} Bajo derivadas segundas continuas y acotadas, se garantiza que \(\liminf_{k\to\infty}|f'(x_k)| = 0\) desde cualquier punto inicial. La condición sobre \(\rho\) previene el incremento del valor funcional.
        \\[0.35em]
        \textbf{Convergencia local:} Si la matriz hessiana (o \(f''\)) en \(x^{*}\) es estrictamente positiva, el modelo aproxima cuadráticamente la función, \(\rho \to 1\), la restricción deja de estar activa y el régimen pasa a ser de Newton con convergencia cuadrática: \(|\varepsilon_{k+1}| = \mathcal{O}(|\varepsilon_k|^{2})\).
    \end{alertblock}

    \vspace{0.3em}
    \footnotesize
    \begin{itemize}
        \item \textbf{Costo computacional por rechazo:} Requiere únicamente una evaluación adicional de la función objetivo, conservando la propiedad de monotonicidad \(f(x_{k+1}) \le f(x_k)\).
        \item \textbf{Estabilidad frente a Newton:} Evita la dispersión ante singularidades o mal condicionamiento de la segunda derivada.
        \item \textbf{Curvatura indefinida o negativa:} Ante puntos de inflexión o sillas (\(f'' < 0\)), la restricción de norma mantiene el iterado acotado sin divergir.
    \end{itemize}
\end{frame}
