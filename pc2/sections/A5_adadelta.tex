\subsection{Adadelta}

\begin{frame}{Adadelta: Explicación Gráfica}
    \begin{block}{Concepto principal}
        [Explica la eliminación de la tasa de aprendizaje explícita usando la curvatura empírica y coherencia de unidades].
    \end{block}
    \begin{itemize}
        \item {[Idea gráfica 1: Relación entre la escala de actualización \(\Delta x\) y el gradiente \(g\)].}
        \item {[Idea gráfica 2: Robustez ante cambios de escala de la función].}
    \end{itemize}
\end{frame}

\begin{frame}{Adadelta: Pseudocódigo}
    \begin{algorithmic}[1]
        \State \textbf{Input:} {[Factor de decaimiento \(\rho\), constante \(\epsilon\), punto \(x\)]}
        \State Inicializar acumuladores: \(E[g^2] \leftarrow 0\), \(E[\Delta x^2] \leftarrow 0\)
        \While{[Condición de parada]}
            \State Calcular gradiente \(g\)
            \State Actualizar \(E[g^2]\) con promedio exponencial
            \State Calcular paso \(\Delta x\) en función de \(\text{RMS}[\Delta x]\) y \(\text{RMS}[g]\)
            \State Actualizar \(E[\Delta x^2]\) y posición \(x \leftarrow x + \Delta x\)
        \EndWhile
        \State \textbf{Return} {[Punto óptimo]}
    \end{algorithmic}
\end{frame}
