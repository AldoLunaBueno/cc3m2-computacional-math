% 2. Pseudocódigo

\begin{frame}{Conjugate Gradient: Pseudocódigo}
    \scriptsize
    \begin{algorithmic}[1]
        \Require $f$, $\nabla f$, $x^{(1)}$, tolerancia $\epsilon$, $k_{\max}$
        \State $g \gets \nabla f(x)$; \quad $d \gets -g$ \Comment{(5.10)}
        \For{$k = 1, \dots, k_{\max}$}
            \State $\alpha \gets \arg\min_{\alpha} f(x + \alpha d)$ \Comment{búsqueda en línea}
            \State $x \gets x + \alpha\, d$ \Comment{(5.11)}
            \State $g' \gets \nabla f(x)$
            \State \textbf{if} $\lVert g' \rVert < \epsilon$ \textbf{then break}
            \State $\beta \gets \max\bigl(0,\ g'^{\top}(g' - g)\,/\,g^{\top} g\bigr)$ \Comment{Polak--Ribière (5.18)--(5.19)}
            \State $d \gets -g' + \beta\, d$ \Comment{(5.12)}
            \State $g \gets g'$
        \EndFor
        \State \Return $x$
    \end{algorithmic}

    \vspace{0.8em}
    \begin{block}{Paso exacto en cuadráticas (línea 3)}
        \centering\footnotesize
        $\alpha = -\dfrac{d^\top(Ax+b)}{d^\top A d}$ \qquad (Ejemplo~5.1)
    \end{block}
    \reflibro{Algoritmo~5.2, Ejemplo~5.1, Ec.~(5.10)--(5.19).}
\end{frame}

\begin{frame}[fragile]{Conjugate Gradient: Código}
\begin{lstlisting}[language=Python]
import numpy as np
from scipy.optimize import minimize_scalar
def line_search(f, x, d):            # alpha = argmin f(x + alpha*d)
    alpha = minimize_scalar(lambda a: f(x + a*d)).x
    return x + alpha*d

def conjugate_gradient(f, grad, x, k_max=100, eps=1e-6):
    g = grad(x)
    d = -g                                  # (5.10) direccion inicial
    for k in range(1, k_max + 1):
        x = line_search(f, x, d)            # (5.11) x <- x + alpha*d
        g_new = grad(x)
        if np.linalg.norm(g_new) < eps:     # criterio de parada
            break
        beta = max(0.0, g_new @ (g_new - g) / (g @ g))  # (5.18)-(5.19)
        d = -g_new + beta*d                 # (5.12) nueva direccion
        g = g_new
    return x, k

# Ejemplo: f(x1, x2) = x1^2 + x1*x2 + x2^2 + 5
f    = lambda x: x[0]**2 + x[0]*x[1] + x[1]**2 + 5
grad = lambda x: np.array([2*x[0] + x[1], x[0] + 2*x[1]])
conjugate_gradient(f, grad, np.array([2.0, 0.0]))    # -> ([0., 0.], 2)
# Rosenbrock (a=1, b=5) del Apendice B.6
ros  = lambda x: (1 - x[0])**2 + 5*(x[1] - x[0]**2)**2
gros = lambda x: np.array([-2*(1 - x[0]) - 20*x[0]*(x[1] - x[0]**2),
                           10*(x[1] - x[0]**2)])
conjugate_gradient(ros, gros, np.array([-1.3, -1.6]))  # -> ([1., 1.], 9)
\end{lstlisting}
\end{frame}
