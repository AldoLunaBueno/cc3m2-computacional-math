\begin{frame}{Hypergradient Descent: Pseudocódigo}
    \begin{algorithmic}[1]
        \State \textbf{Input:} {[Tasa inicial \(\alpha_0\), hiper-tasa \(\mu\), posición \(x_0\)]}
        \State Calcular gradiente inicial \(g_0\)
        \While{[Condición de parada]}
            \State Actualizar posición usando \(\alpha_{t-1}\) y \(g_{t-1}\)
            \State Calcular nuevo gradiente \(g_t\)
            \State Actualizar tasa de aprendizaje: \(\alpha_t \leftarrow \alpha_{t-1} + \mu \langle g_t, g_{t-1} \rangle\)
        \EndWhile
        \State \textbf{Return} {[Punto óptimo]}
    \end{algorithmic}
\end{frame}
