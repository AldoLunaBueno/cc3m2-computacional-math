% 2. Pseudocódigo
% ---------------------------------------------------------------
\begin{frame}{Adagrad: Pseudocódigo}
    \scriptsize
    \begin{algorithmic}[1]
        \Require $\nabla f$, $x^{(1)}$, $\alpha$, $\epsilon\approx10^{-8}$, tolerancia $\delta$, $k_{\max}$
        \State $s \gets 0$
        \For{$k = 1, \dots, k_{\max}$}
            \State $g \gets \nabla f(x)$
            \State \textbf{if} $\lVert g \rVert < \delta$ \textbf{then break}
            \State $s \gets s + g \odot g$ \Comment{(5.26)}
            \State $x \gets x - \alpha\, g \oslash \bigl(\epsilon + \sqrt{s}\bigr)$ \Comment{(5.24)--(5.25)}
        \EndFor
        \State \Return $x$
    \end{algorithmic}

    \vspace{1em}
    \small
    \begin{itemize}
        \item $\odot$, $\oslash$, $\sqrt{\cdot}$: \textbf{componente a componente}
        \item $k=1$: cada componente se mueve exactamente $\alpha$
        \item Por defecto (libro): $\alpha=0.01$
    \end{itemize}
    \reflibro{Algoritmo~5.5, Ec.~(5.24)--(5.26).}
\end{frame}

\begin{frame}[fragile]{Adagrad: Código}
\begin{lstlisting}[language=Python]
import numpy as np

def adagrad(grad, x, alpha=0.01, eps=1e-8, k_max=1000, tol=1e-6):
    s = np.zeros_like(x)                      # suma de g^2 por componente
    for k in range(1, k_max + 1):
        g = grad(x)
        if np.linalg.norm(g) < tol:           # criterio de parada
            break
        s = s + g*g                           # (5.26) acumulador
        x = x - alpha*g / (eps + np.sqrt(s))  # (5.24)-(5.25)
    return x, k

# Ejemplo: f(x) = x1^2 + 10*x2^2   ->   grad f = (2*x1, 20*x2)
grad = lambda x: np.array([2*x[0], 20*x[1]])
x0 = np.array([-3.0, 1.0])
adagrad(grad, x0, alpha=0.5, k_max=3)       # x(4) = [-1.9362, 0.1564]
adagrad(grad, x0, alpha=0.5,  k_max=10**5)  # -> [0, 0] en k = 191
adagrad(grad, x0, alpha=1.0,  k_max=10**5)  # k = 58
adagrad(grad, x0, alpha=5.0,  k_max=10**5)  # k = 17
adagrad(grad, x0, alpha=0.01, k_max=10**5)  # no llega: x1 = -0.0224
\end{lstlisting}
\end{frame}

            \begin{alertblock}{El paso efectivo solo baja}
                \footnotesize
                $s_i$ no decrece $\Rightarrow$ $\dfrac{\alpha}{\epsilon+\sqrt{s_i}}$ no crece
            \end{alertblock}
            \small
            \begin{itemize}
                \item $g$ no decae: paso $\sim 1/\sqrt{k}$ $\Rightarrow$ se \textbf{congela}\\
