\begin{frame}{RMSProp: Explicación Gráfica}

    \end{block}
    \begin{itemize}
        \item {[Idea gráfica 1: Ventana móvil de memoria reciente en lugar de suma histórica infinita].}
        \item {[Idea gráfica 2: Estabilización de trayectorias en valles alargados].}
    \end{itemize}
\end{frame}

\begin{frame}{RMSProp: Pseudocódigo}
    \begin{algorithmic}[1]
        \State \textbf{Input:} {[Tasa \(\alpha\), factor de decaimiento \(\gamma\), \(\epsilon\), punto \(x\)]}
        \State Inicializar promedio móvil \(v \leftarrow 0\)
        \While{[Condición de parada]}
            \State Calcular gradiente \(g\)
            \State Actualizar promedio exponencial: \(v \leftarrow \gamma v + (1-\gamma) g \odot g\)
            \State Actualizar parámetros: \(x \leftarrow x - \frac{\alpha}{\sqrt{v + \epsilon}} \odot g\)
        \EndWhile
        \State \textbf{Return} {[Punto óptimo]}
    \end{algorithmic}
\end{frame}
