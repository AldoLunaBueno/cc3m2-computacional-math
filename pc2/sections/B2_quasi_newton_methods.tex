\subsection{Quasi-Newton Methods}

\begin{frame}{Quasi-Newton Methods: Explicación Gráfica}
    \begin{block}{Concepto principal}
        [Explica la aproximación iterativa de la matriz inversa del Hessiano \(H^{-1}\) usando solo información de gradientes].
    \end{block}
    \begin{itemize}
        \item {[Idea gráfica 1: Modelo cuadrático local que se ajusta iterativamente].}
        \item {[Idea gráfica 2: Corrección secante matricial en varias dimensiones].}
    \end{itemize}
\end{frame}

\begin{frame}{Quasi-Newton Methods: Pseudocódigo}
    \begin{algorithmic}[1]
        \State \textbf{Input:} {[Gradiente \(\nabla f\), matriz inicial \(B_0\) (o \(H_0^{-1}\)), punto \(x_0\)]}
        \While{[Condición de parada]}
            \State Calcular dirección de búsqueda: \(d \leftarrow -H \nabla f(x)\)
            \State Realizar búsqueda de línea para obtener \(\alpha\)
            \State Calcular diferencias de posiciones \(s\) y gradientes \(y\)
            \State Actualizar matriz aproximada (ej. fórmula BFGS o DFP)
            \State \(x \leftarrow x + \alpha d\)
        \EndWhile
        \State \textbf{Return} {[Punto óptimo]}
    \end{algorithmic}
\end{frame}
