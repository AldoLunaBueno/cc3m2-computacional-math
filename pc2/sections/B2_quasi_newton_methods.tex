            \end{tikzpicture}
        \end{column}
        
        \begin{column}{0.54\textwidth}
            \scriptsize
            \begin{block}{La Limitación del Gradiente}
                El gradiente puro es perpendicular a las curvas de nivel. En valles estrechos, esto obliga a dar pasos en direcciones subóptimas, ignorando el centro del valle.
            \end{block}
            
            \vspace{0.3cm}
            \begin{itemize}
                \setlength{\itemsep}{0.8em}
                \item \textbf{La Solución de Newton:} Usar la matriz Hessiana para capturar la curvatura del valle, pero calcular su inversa es computacionalmente prohibitivo.
                \item \textbf{La Estrategia Quasi-Newton:} Aproximar la inversa del Hessiano iterativamente observando cómo cambia el gradiente paso a paso.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

        \scriptsize
        \begin{column}{0.54\textwidth}
            \begin{block}{Algoritmos BFGS y L-BFGS}
                Los algoritmos de la familia Quasi-Newton construyen iterativamente la topología del espacio sin segundas derivadas explícitas.
            \end{block}
            
            \vspace{0.2cm}
            \begin{itemize}
                \setlength{\itemsep}{0.8em}
                \item \textbf{Convergencia Superlineal:} Alcanza el mínimo en una fracción de las iteraciones que requiere el Gradiente Descendente, como se aprecia en la rápida caída del error en el diagrama de barras.
                \item \textbf{Costo de Memoria:} La matriz densa de $B^{-1}$ exige $O(n^2)$ en memoria. 
                \item \textbf{L-BFGS (Limited-memory):} Resuelve el problema de memoria guardando solo los últimos $m$ vectores espaciales en lugar de toda la matriz, ideal para Deep Learning.
            \end{itemize}
        \end{column}
    \end{columns}

\begin{frame}{Quasi-Newton Methods: Pseudocódigo}
    \begin{algorithmic}[1]
        \State \textbf{Input:} {[Gradiente \(\nabla f\), matriz inicial \(B_0\) (o \(H_0^{-1}\)), punto \(x_0\)]}
        \While{[Condición de parada]}
            \State Calcular dirección de búsqueda: \(d \leftarrow -H \nabla f(x)\)
            \State Realizar búsqueda de línea para obtener \(\alpha\)
            \State Calcular diferencias de posiciones \(s\) y gradientes \(y\)
            \State Actualizar matriz aproximada (ej. fórmula BFGS o DFP)
            \State \(x \leftarrow x + \alpha d\)
        \EndWhile
        \State \textbf{Return} {[Punto óptimo]}
    \end{algorithmic}
\end{frame}
